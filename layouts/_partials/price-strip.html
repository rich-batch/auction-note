{{- /* 가격 비교 그래프: 가격 축 위에 출처별 한 줄씩, 비교 사례를 점으로 찍고 출처별 평균을 짧은 막대로, 이 물건의 최저가(실선, 위 라벨)와 낙찰가(점선, 아래 라벨)를 세로선으로 그린다. 사용: partial "price-strip.html" $c — computed.price_gap(by_source)이 있을 때만 그린다. 출처 구분은 줄 이름이 맡아서 점 색은 하나로 둔다. 같은 숫자는 바로 아래 표에도 있다(표 보기). */ -}} {{- $c := . -}} {{- $groups := $c.computed.price_gap.by_source -}} {{- $trades := $c.market.trades | default slice -}} {{- $min := float $c.sale.minimum -}} {{- $bid := 0.0 -}} {{- with $c.result }}{{ with .winning_bid }}{{ $bid = float . }}{{ end }}{{ end -}} {{- /* 축 범위: 모든 값(사례·최저가·낙찰가)을 담는 깔끔한 눈금 간격을 고른다 */ -}} {{- $vals := slice $min -}} {{- if gt $bid 0.0 }}{{ $vals = $vals | append $bid }}{{ end -}} {{- range $trades }}{{ $vals = $vals | append (float .price) }}{{ end -}} {{- $rawLo := math.Min $vals -}} {{- $rawHi := math.Max $vals -}} {{- $step := 1e9 -}} {{- range slice 1e6 2e6 5e6 1e7 2e7 5e7 1e8 2e8 5e8 1e9 -}} {{- if and (eq $step 1e9) (le (div (sub $rawHi $rawLo) .) 5.0) }}{{ $step = . }}{{ end -}} {{- end -}} {{- $lo := mul (math.Floor (div $rawLo $step)) $step -}} {{- $hi := mul (math.Ceil (div $rawHi $step)) $step -}} {{- if le $hi $lo }}{{ $hi = add $lo $step }}{{ end -}} {{- $span := sub $hi $lo -}} {{- $nTicks := int (math.Round (div $span $step)) -}}
